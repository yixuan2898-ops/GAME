    <script>
        // 词汇库
        const vocabulary = {
            school: [
                { en: "Pencil", zh: "铅笔", emoji: "✏️", sentence: "I write with a pencil." },
                { en: "Book", zh: "书本", emoji: "📖", sentence: "She is reading a book." },
                { en: "Desk", zh: "书桌", emoji: "🪑", sentence: "My desk is clean." },
                { en: "Eraser", zh: "橡皮", emoji: "🧽", sentence: "I need an eraser." },
                { en: "Ruler", zh: "尺子", emoji: "📏", sentence: "Use a ruler to draw a line." },
                { en: "Bag", zh: "书包", emoji: "🎒", sentence: "My bag is heavy." }
            ],
            food: [
                { en: "Apple", zh: "苹果", emoji: "🍎", sentence: "An apple a day." },
                { en: "Bread", zh: "面包", emoji: "🍞", sentence: "I like eating bread." },
                { en: "Milk", zh: "牛奶", emoji: "🥛", sentence: "Drink your milk." },
                { en: "Egg", zh: "鸡蛋", emoji: "🥚", sentence: "I have an egg for breakfast." },
                { en: "Cake", zh: "蛋糕", emoji: "🍰", sentence: "Happy birthday cake!" },
                { en: "Juice", zh: "果汁", emoji: "🧃", sentence: "Orange juice is sweet." }
            ],
            body: [
                { en: "Eye", zh: "眼睛", emoji: "👁️", sentence: "Close your eyes." },
                { en: "Ear", zh: "耳朵", emoji: "👂", sentence: "I hear with my ears." },
                { en: "Nose", zh: "鼻子", emoji: "👃", sentence: "Touch your nose." },
                { en: "Hand", zh: "手", emoji: "🖐️", sentence: "Wash your hands." },
                { en: "Foot", zh: "脚", emoji: "🦶", sentence: "My left foot." },
                { en: "Tooth", zh: "牙齿", emoji: "🦷", sentence: "Brush your teeth." }
            ]
        };

        let currentTheme = 'school';
        let cards = [];
        let flippedCards = [];
        let matchedPairs = 0;
        let score = 0;
        let combo = 0;
        let timerInterval;
        let seconds = 0;
        let isLocked = false;

        const synth = window.speechSynthesis;

        // 语音朗读
        function speak(text) {
            const utterance = new SpeechSynthesisUtterance(text);
            utterance.lang = 'en-US';
            utterance.rate = 0.9; // 稍微放慢语速适合儿童
            synth.speak(utterance);
        }

        // 切换界面
        function showScreen(screenId) {
            document.querySelectorAll('body > div').forEach(div => div.classList.remove('active'));
            document.getElementById(screenId).classList.add('active');
            if(screenId !== 'game-screen') clearInterval(timerInterval);
        }

        function goHome() {
            showScreen('home-screen');
        }

        // 启动游戏
        function startGame() {
            currentTheme = document.getElementById('theme-select').value;
            score = 0;
            combo = 0;
            seconds = 0;
            matchedPairs = 0;
            flippedCards = [];
            isLocked = false;
            
            updateStats();
            document.getElementById('timer').innerText = "00:00";
            document.getElementById('teaching-board').innerText = "点击卡片配对哦！";
            
            // 准备卡片 (每个单词生成两张一样的卡片用于消消乐)
            const selectedWords = vocabulary[currentTheme];
            cards = [...selectedWords, ...selectedWords];
            cards.sort(() => Math.random() - 0.5); // 洗牌
            
            renderGrid();
            showScreen('game-screen');
            
            timerInterval = setInterval(() => {
                seconds++;
                const m = Math.floor(seconds / 60).toString().padStart(2, '0');
                const s = (seconds % 60).toString().padStart(2, '0');
                document.getElementById('timer').innerText = `${m}:${s}`;
            }, 1000);
        }

        // 渲染网格
        function renderGrid() {
            const grid = document.getElementById('game-grid');
            grid.innerHTML = '';
            
            cards.forEach((wordObj, index) => {
                const card = document.createElement('div');
                card.classList.add('card');
                card.dataset.index = index;
                card.dataset.word = wordObj.en;
                
                card.innerHTML = `
                    <div class="card-emoji">${wordObj.emoji}</div>
                    <div class="card-word">${wordObj.en}</div>
                `;
                
                card.addEventListener('click', () => flipCard(card, wordObj));
                grid.appendChild(card);
            });
        }

        // 翻卡逻辑
        function flipCard(card, wordObj) {
            if (isLocked || card.classList.contains('flipped') || card.classList.contains('matched')) return;

            card.classList.add('flipped');
            flippedCards.push(card);
            
            // 播放声音并显示教学信息
            speak(wordObj.en);
            document.getElementById('teaching-board').innerHTML = `
                ${wordObj.en} (${wordObj.zh}) <br> <span style="font-size:14px; color:#555;">${wordObj.sentence}</span>
            `;

            if (flippedCards.length === 2) {
                checkMatch();
            }
        }

        // 检查匹配
        function checkMatch() {
            isLocked = true;
            const [card1, card2] = flippedCards;
            
            if (card1.dataset.word === card2.dataset.word) {
                // 成功配对
                setTimeout(() => {
                    card1.classList.add('matched');
                    card2.classList.add('matched');
                    matchedPairs++;
                    combo++;
                    score += 10 * combo;
                    updateStats();
                    showFeedback(true);
                    
                    if (matchedPairs === vocabulary[currentTheme].length) {
                        setTimeout(endGame, 800);
                    }
                    resetBoard();
                }, 800);
            } else {
                // 配对失败
                setTimeout(() => {
                    card1.classList.remove('flipped');
                    card2.classList.remove('flipped');
                    combo = 0;
                    updateStats();
                    showFeedback(false);
                    resetBoard();
                }, 1000);
            }
        }

        function resetBoard() {
            flippedCards = [];
            isLocked = false;
        }

        function updateStats() {
            document.getElementById('score').innerText = score;
            document.getElementById('combo').innerText = combo;
        }

        // 浮动鼓励提示
        function showFeedback(isSuccess) {
            const feedbackEl = document.getElementById('feedback');
            feedbackEl.classList.remove('show-feedback');
            
            if (isSuccess) {
                const msgs = ["Great Job!", "Excellent!", "Awesome!", "Combo!"];
                feedbackEl.innerText = msgs[Math.floor(Math.random() * msgs.length)];
                feedbackEl.style.color = "#4CAF50";
            } else {
                feedbackEl.innerText = "Try Again! 🦌";
                feedbackEl.style.color = "#FF9800";
            }
            
            // Trigger reflow
            void feedbackEl.offsetWidth; 
            feedbackEl.classList.add('show-feedback');
        }

        // 游戏结束
        function endGame() {
            clearInterval(timerInterval);
            document.getElementById('final-score').innerText = score;
            
            const list = document.getElementById('learned-words');
            list.innerHTML = '';
            vocabulary[currentTheme].forEach(word => {
                const li = document.createElement('li');
                li.innerHTML = `✅ <b>${word.en}</b> - ${word.zh}`;
                li.style.marginBottom = "5px";
                list.appendChild(li);
            });
            
            showScreen('end-screen');
            speak("Awesome job! You are a superstar!");
        }
    </script>
